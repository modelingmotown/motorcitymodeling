# 23. Regression Models (SPSS: REGRESSION /DEPENDENT=LN_tasp /METHOD=BACKWARD ...)
# The main regression model:
# Note: Ensure all variables used in the model exist in your 'df_processed' data frame.
# If some variables are not created due to previous comments in SPSS, you'll need to adjust.
# Also, R's lm() doesn't directly support backward selection as an argument.

# The second regression in the file (SPSS: REGRESSION /DEPENDENT=LN_tasp /METHOD=BACKWARD ...)
# This model uses the time-adjusted sale price (LN_tasp).
# lnsbldsfINT was part of the original SPSS REGRESSION

lm_model_tasp <- lm(ln_tasp ~ ln_landratio + lnsbldsfint + arterial +
                      single_family1story + half_duplex1story + half_duplex2story +
                      smallest + small + large + largest + two_plus_baths + powder_room +
